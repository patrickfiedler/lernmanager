function toggleSubtask(position, checked) {
    const csrfToken = document.querySelector('meta[name="csrf-token"]').content;

    // Disable checkbox during request
    const checkboxes = document.querySelectorAll('.subtask-checkbox');
    checkboxes.forEach(cb => cb.disabled = true);

    fetch(`/schueler/thema/${topicSlug}/aufgabe/${position}`, {
        method: 'POST',
        headers: {
            'Content-Type': 'application/json',
            'X-CSRFToken': csrfToken
        },
        body: JSON.stringify({erledigt: checked})
    })
    .then(response => {
        if (!response.ok) {
            throw new Error(`HTTP error! status: ${response.status}`);
        }
        return response.json();
    })
    .then(data => {
        if (data.show_quiz && data.quiz_url) {
            window.location.href = data.quiz_url;
            return;
        }

        if (checked) {
            const successBanner = document.getElementById('success-banner');
            if (successBanner) {
                successBanner.style.display = 'flex';
            }

            // Update progress dot
            const currentDot = document.querySelector('.progress-dots .dot.current');
            if (currentDot) {
                currentDot.classList.add('completed');
            }

            // Update quiz dot so goToNextSubtask() knows the quiz is now available
            const quizDot = document.querySelector(`.dot-subtask-quiz[data-quiz-position="${position}"]`);
            if (quizDot && quizDot.dataset.quizPassed !== 'true') {
                quizDot.dataset.quizAvailable = 'true';
                quizDot.classList.add('available');
            }

            updateProgressText();
        } else {
            // Unchecking - reload page to reset state
            setTimeout(() => {
                window.location.href = window.location.href;
            }, 100);
        }
    })
    .catch(err => {
        console.error('Error toggling subtask:', err);
        checkboxes.forEach(cb => cb.disabled = false);
        alert(`Fehler beim Speichern: ${err.message}\nBitte versuche es erneut oder lade die Seite neu.`);
    });
}

function updateProgressText() {
    const allDots = document.querySelectorAll('.progress-dots .dot.dot-subtask');
    const requiredDots = Array.from(allDots).filter(d => d.dataset.required !== 'false' && d.dataset.isIntro !== 'true');
    const completedRequired = requiredDots.filter(d => d.classList.contains('completed'));
    const progressText = document.getElementById('progress-text');

    if (progressText) {
        progressText.textContent = `${completedRequired.length} von ${requiredDots.length} Aufgaben erledigt`;
    }
}

function goToNextSubtask(currentIndex) {
    const position = currentIndex + 1; // 1-based
    // Check if current task has an unpassed quiz → go there first
    const quizDot = document.querySelector(`.dot-subtask-quiz[data-quiz-position="${position}"]`);
    if (quizDot && quizDot.dataset.quizPassed !== 'true' && quizDot.dataset.quizAvailable === 'true') {
        navigateToSubtaskQuiz(position, false, true);
        return;
    }
    if (currentIndex < subtaskCount - 1) {
        const url = new URL(window.location);
        url.searchParams.set('aufgabe', currentIndex + 2);
        window.location.href = url.toString();
    } else {
        const topicQuizDot = document.querySelector('.dot-quiz');
        if (topicQuizDot) {
            navigateToQuiz(false);
        }
    }
}

function goToPrevSubtask(currentIndex) {
    if (currentIndex > 0) {
        const url = new URL(window.location);
        url.searchParams.set('aufgabe', currentIndex);
        window.location.href = url.toString();
    }
}

function navigateToSubtask(position) {
    const url = new URL(window.location);
    url.searchParams.set('aufgabe', position);
    window.location.href = url.toString();
}

function navigateToQuiz(passed) {
    if (!passed && gatePosition && !gatePassed) {
        const gateCard = document.getElementById('gate-card');
        if (gateCard) {
            gateCard.scrollIntoView({behavior: 'smooth'});
        } else {
            navigateToSubtask(gatePosition);
        }
        return;
    }
    if (passed) {
        window.location.href = `/schueler/thema/${topicSlug}/quiz-ergebnis`;
    } else {
        window.location.href = `/schueler/thema/${topicSlug}/quiz`;
    }
}

function navigateToSubtaskQuiz(position, passed, available) {
    if (passed) {
        window.location.href = `/schueler/thema/${topicSlug}/aufgabe-${position}/quiz-ergebnis`;
    } else if (available) {
        window.location.href = `/schueler/thema/${topicSlug}/aufgabe-${position}/quiz`;
    }
    // locked (not available) — do nothing
}

// Keyboard navigation for all dots (WCAG 2.1)
function handleDotKeydown(event) {
    if (event.key === 'Enter' || event.key === ' ') {
        event.preventDefault();
        event.target.click();
    } else if (event.key === 'ArrowLeft' || event.key === 'ArrowRight') {
        event.preventDefault();
        const dots = Array.from(document.querySelectorAll('.progress-dots .dot'));
        const currentIndex = dots.indexOf(event.target);

        let nextIndex;
        if (event.key === 'ArrowLeft') {
            nextIndex = currentIndex > 0 ? currentIndex - 1 : dots.length - 1;
        } else {
            nextIndex = currentIndex < dots.length - 1 ? currentIndex + 1 : 0;
        }

        dots[nextIndex].focus();
    }
}

let artifactPreviewText = '';
let artifactPreviewFilename = '';

function handleArtifactUpload(position) {
    const fileInput = document.getElementById('artifact-file-input');
    const file = fileInput.files[0];
    if (!file) return;

    document.getElementById('artifact-filename').textContent = file.name;
    const kwWarning = document.getElementById('artifact-keyword-warning');
    if (kwWarning) {
        if (artifactKeyword && !file.name.toLowerCase().includes(artifactKeyword.toLowerCase())) {
            kwWarning.textContent = `⚠️ Dateiname sollte „${artifactKeyword}“ enthalten.`;
            kwWarning.style.display = '';
        } else {
            kwWarning.style.display = 'none';
        }
    }
    document.getElementById('artifact-status').textContent = 'Wird gelesen …';
    const feedbackBtn = document.getElementById('artifact-feedback-btn');
    feedbackBtn.style.display = 'none';
    feedbackBtn.textContent = 'KI-Check starten →';
    feedbackBtn.disabled = false;
    document.getElementById('artifact-checklist').style.display = 'none';
    document.getElementById('artifact-preview-details').style.display = 'none';

    const csrfToken = document.querySelector('meta[name="csrf-token"]').content;
    const formData = new FormData();
    formData.append('file', file);

    fetch(`/schueler/thema/${topicSlug}/aufgabe-${position}/artefakt/vorschau`, {
        method: 'POST',
        headers: {'X-CSRFToken': csrfToken},
        body: formData,
    })
    .then(r => r.json())
    .then(data => {
        if (data.error) {
            document.getElementById('artifact-status').textContent = '⚠️ ' + data.error;
            return;
        }
        artifactPreviewText = data.preview_text;
        artifactPreviewFilename = data.filename;
        document.getElementById('artifact-preview-text').textContent = data.preview_text;
        document.getElementById('artifact-preview-filename').textContent = data.filename;
        document.getElementById('artifact-preview-details').style.display = '';
        document.getElementById('artifact-preview-details').open = true;
        document.getElementById('artifact-feedback-btn').style.display = '';
        document.getElementById('artifact-status').textContent = '';
        if (data.file_url) {
            const linkEl = document.getElementById('artifact-file-link');
            document.getElementById('artifact-file-link-a').href = data.file_url;
            document.getElementById('artifact-file-link-a').textContent = data.filename;
            linkEl.style.display = '';
        }
    })
    .catch(() => {
        document.getElementById('artifact-status').textContent = '⚠️ Fehler beim Lesen der Datei.';
    });
}

function requestArtifactFeedback(position) {
    const btn = document.getElementById('artifact-feedback-btn');
    const statusEl = document.getElementById('artifact-status');
    btn.disabled = true;
    btn.innerHTML = '' +
                    '🤖 KI prüft dein Dokument …';
    statusEl.textContent = '';

    // Server budget for a full checklist is 60s (config.LLM_ARTIFACT_TIMEOUT), so
    // the client window sits above it -- but it must exist: without one, a hung
    // request fires neither .then nor .catch and the button stays disabled forever.
    LLMButton.postJSON(`/schueler/thema/${topicSlug}/aufgabe-${position}/artefakt/feedback`,
        {text: artifactPreviewText, filename: artifactPreviewFilename},
        {timeoutMs: 90000})
    .then(data => {
        statusEl.textContent = '';
        btn.textContent = 'Erneut prüfen →';
        btn.disabled = false;
        if (!data.ok) {
            btn.textContent = 'KI-Check starten →';
            statusEl.textContent = '⚠️ ' + data.message;
            return;
        }
        if (data.error) {
            statusEl.textContent = '⚠️ ' + data.error;
            return;
        }
        if (data.rate_limited) {
            statusEl.innerHTML = '⏰ Du hast heute schon 10 Mal geprüft. Warte eine Stunde — dann kannst du wieder prüfen.';
        } else if (data.llm_disabled) {
            statusEl.textContent = 'ℹ️ KI-Feedback ist für diese Klasse nicht aktiviert.';
        } else if (data.checks_remaining !== undefined && data.checks_remaining <= 3 && data.checks_remaining > 0) {
            statusEl.textContent = `Noch ${data.checks_remaining} Check${data.checks_remaining === 1 ? '' : 's'} in dieser Stunde übrig.`;
        }
        // Filename check can still produce feedback even when LLM is off/rate-limited;
        // only show the "couldn't evaluate" error when that's not the explanation.
        if ((data.feedback && data.feedback.length) || (!data.rate_limited && !data.llm_disabled)) {
            renderArtifactChecklist(data.feedback, data.extracted_text);
        } else {
            document.getElementById('artifact-checklist').style.display = 'none';
        }
    });
}

function renderFeedbackSplit(feedback) {
    if (!feedback || !feedback.length) return '';
    const good = feedback.filter(item => item.passed);
    const bad = feedback.filter(item => !item.passed);
    const row = item => `

        ${item.passed ? '✅' : (item.severity === 'warn' ? '⚠️' : '❌')}
        

            
${escapeHtml(item.criterion)}

            ${item.note ? `
💬 ${escapeHtml(item.note)}${item.source !== 'deterministic' ? ' ✨ KI' : ''}
` : ''}
        

    
`;
    // The two columns are not mirror images, so their empty states are not either.
    // "Hier gibt's noch was zu tun" is meant to empty out -- when it does the column
    // goes away entirely and the green panel takes the full width; that structural
    // change IS the success signal, and a stronger one than an empty red column.
    // "Gut so" empty means "not yet", a starting state, so it stays and says where to
    // look. Both used to render "-", and students read the empty red half as
    // "something is missing but the site won't tell me what" -- an absence of problems
    // and an absence of information had the same glyph.
    const BAD_LABEL = "Hier gibt's noch was zu tun";
    const col = (title, items, cls) => `

        ${title ? `
${title}
` : ''}
        ${items.length ? items.map(row).join('')
                       : `
Mögliche Verbesserungen stehen unter „${BAD_LABEL}“.
`}
    
`;
    const header = feedback.some(item => item.source !== 'deterministic') ? 'KI-Feedback:' : 'Prüfung:';
    // No column title in single-column mode: "✅ Gut so" labels a column against its
    // counterpart, and there is no counterpart left to contrast it with.
    const allDone = bad.length === 0;
    const cols = allDone ? col('', good, 'good')
                         : col('✅ Gut so', good, 'good') + col('❌ ' + BAD_LABEL, bad, 'bad');
    return `
${header}

        
${cols}

        
Note wird von der Lehrkraft gesetzt.
`;
}

function renderArtifactChecklist(feedback, extractedText) {
    const container = document.getElementById('artifact-checklist');
    if (!feedback || feedback.length === 0) {
        container.innerHTML = '
⚠️ KI konnte die Datei nicht auswerten. Bitte nochmal versuchen.
';
        container.style.display = '';
        return;
    }
    const transparencyBlock = extractedText ? `
        

            ✨ Was die KI gesehen hat
            
${escapeHtml(extractedText)}

        
` : '';
    container.innerHTML = renderFeedbackSplit(feedback) + transparencyBlock;
    container.style.display = '';
}

function escapeHtml(str) {
    return str.replace(/&/g,'&').replace(//g,'>').replace(/"/g,'"');
}

// Structural gate results arrive as bare strings; reshape them into the shared
// feedback contract so they render in the same split table as LLM criteria.
// Warnings count as "noch was zu tun" -- the split view has no third column.
function gateFeedbackItems(matches, details, warnings, filenameItem) {
    const items = [];
    if (filenameItem) {
        items.push({ criterion: filenameItem.text, passed: filenameItem.type === 'pass',
                     severity: filenameItem.type === 'warn' ? 'warn' : undefined,
                     source: 'deterministic' });
    }
    (matches || []).forEach(m => items.push({ criterion: m, passed: true, source: 'deterministic' }));
    (warnings || []).forEach(w => items.push({ criterion: w, passed: false, severity: 'warn',
                                               source: 'deterministic' }));
    (details || []).forEach(d => items.push({ criterion: d, passed: false, source: 'deterministic' }));
    return items;
}

// One list, two sources: deterministic gate hits first, LLM criteria after.
// These used to be an either/or -- and llm_feedback always won, because the
// deterministic filename check runs on every passed gate and makes it non-empty.
// A student who got everything right saw an empty "Gut so" column.
// The client's keyword check is only a fallback: when the server ran the exact
// expected_filename check, drop it, or the card shows two filename lines that can
// contradict each other (keyword present, exact name wrong).
// A failing re-check retracts the pass server-side (app.py), but the card header and
// the completion zone are both server-rendered from the state at page load. Without this
// the student keeps the green passed-header above a fresh list of errors, and a live
// completion checkbox one click from the quiz. Persisting the retraction only fixes the
// NEXT page load; this fixes the page in front of them.
// (Deliberately no UI strings in this comment -- several tests assert on those by
// substring, and a comment quoting them makes the page match when it should not.)
// The card is found from the button rather than by id, so one pair of helpers serves
// both the inline and the capstone gate.
function markGateFailed(card) {
    if (!card) return;
    card.classList.remove('gate-passed');
    card.classList.add('gate-ready');
    const header = card.querySelector('.gate-header');
    if (header) header.textContent = '📎 Datei prüfen';
    const success = card.querySelector('.gate-success-msg');
    if (success) success.remove();
    // .gate-desc is deliberately left alone: the two gates word it differently, and
    // "Neue Version einreichen:" stays true after a failure anyway.
}

// The same staleness in the other direction: a card that loaded unpassed kept its amber
// header after a passing check, so the student read "Datei prüfen" above their own
// success. The result panel already carries the message, so no success line is re-added.
function markGatePassed(card) {
    if (!card) return;
    card.classList.remove('gate-ready');
    card.classList.add('gate-passed');
    const header = card.querySelector('.gate-header');
    if (header) header.textContent = '📎 Abgabe geprüft ✓';
    const desc = card.querySelector('.gate-desc');
    if (desc) desc.textContent = 'Neue Version einreichen:';
}

function allFeedbackItems(data, filenameItem) {
    return gateFeedbackItems(data.matches, data.details, data.warnings,
                             data.filename_checked ? null : filenameItem)
           .concat(data.llm_feedback || []);
}

function openSplitPane(btn) {
    const type = btn.dataset.type;
    const url = type === 'youtube' ? (btn.dataset.embedUrl || btn.dataset.url) : btn.dataset.url;
    const title = btn.dataset.title || 'Material';
    const icons = { pdf: '📕', image: '🖼️', youtube: '▶️' };

    const sidebar = document.getElementById('split-pane-sidebar');
    const frame = document.getElementById('split-viewer-frame');
    const imgWrap = document.getElementById('split-viewer-img-wrap');
    const img = document.getElementById('split-viewer-img');

    document.getElementById('split-sidebar-icon').textContent = icons[type] || '📄';
    document.getElementById('split-sidebar-title').textContent = title;
    document.getElementById('split-zoom-controls').style.display = type === 'image' ? '' : 'none';

    if (type === 'image') {
        frame.style.display = 'none';
        frame.src = '';
        img.style.width = '100%';
        img._zoomLevel = 100;
        img.src = url;
        imgWrap.style.display = '';
    } else {
        imgWrap.style.display = 'none';
        img.src = '';
        frame.src = (type === 'pdf') ? url + '#navpanes=0&pagemode=none' : url;
        frame.style.display = '';
    }

    sidebar.style.display = '';
    document.body.classList.add('split-pane-active');
    updateSidebarTop();
    window.addEventListener('scroll', updateSidebarTop);

    document.querySelectorAll('.btn-split').forEach(function(b) { b.classList.remove('split-active'); });
    btn.classList.add('split-active');
}

function updateSidebarTop() {
    const nav = document.querySelector('nav');
    const top = nav ? Math.max(0, nav.offsetHeight - window.scrollY) : 0;
    document.getElementById('split-pane-sidebar').style.top = top + 'px';
}

function closeSplitPane() {
    const sidebar = document.getElementById('split-pane-sidebar');
    sidebar.style.display = 'none';
    document.getElementById('split-viewer-frame').src = '';
    document.getElementById('split-viewer-img').src = '';
    document.getElementById('split-viewer-img-wrap').style.display = 'none';
    document.body.classList.remove('split-pane-active');
    document.body.style.removeProperty('--split-sidebar-width');
    document.querySelectorAll('.btn-split').forEach(function(b) { b.classList.remove('split-active'); });
    window.removeEventListener('scroll', updateSidebarTop);
}

(function() {
    const handle = document.getElementById('split-resize-handle');
    if (!handle) return;
    let dragging = false;
    handle.addEventListener('mousedown', function(e) {
        dragging = true;
        handle.classList.add('dragging');
        document.body.style.cursor = 'col-resize';
        document.body.style.userSelect = 'none';
        // Prevent iframe from stealing mouse events during drag
        document.getElementById('split-viewer-frame').style.pointerEvents = 'none';
        e.preventDefault();
    });
    document.addEventListener('mousemove', function(e) {
        if (!dragging) return;
        const pct = ((window.innerWidth - e.clientX) / window.innerWidth * 100);
        if (pct >= 20 && pct <= 80) {
            const val = pct.toFixed(1) + '%';
            document.getElementById('split-pane-sidebar').style.width = val;
            document.body.style.setProperty('--split-sidebar-width', val);
        }
    });
    document.addEventListener('mouseup', function() {
        if (!dragging) return;
        dragging = false;
        handle.classList.remove('dragging');
        document.body.style.cursor = '';
        document.body.style.userSelect = '';
        document.getElementById('split-viewer-frame').style.pointerEvents = '';
    });
})();

        // 90s window: the gate may run a full LLM checklist server-side (60s budget).
        // Without a timeout a hung upload fires neither .then nor .catch, leaving the
        // button disabled with no way to retry.
        LLMButton.postForm(submitBtn.dataset.url, formData, { timeoutMs: 90000 })
            .then(data => {

⚠️ ${data.message}

⚠️ ${data.error}

                    submitBtn.disabled = false;
                    submitBtn.textContent = 'Datei prüfen';
                    return;

                // rendered from the previous upload is stale -- drop it rather than

Noch ${data.checks_remaining} KI-Check${data.checks_remaining === 1 ? '' : 's'} übrig.
` : '';
                    markGatePassed(submitBtn.closest('.gate-card'));
                    let html = `
✓ Abgabe geprüft!
`;
                    html += renderFeedbackSplit(allFeedbackItems(data, filenameItem)) + rateNote;
                    html += ``;
                    resultDiv.innerHTML = html;
                    resultDiv.style.display = 'block';
                    submitBtn.style.display = 'none';
                } else {
                    markGateFailed(submitBtn.closest('.gate-card'));

${data.message}

                    resultDiv.innerHTML = html;
                    resultDiv.style.display = 'block';
                    submitBtn.disabled = false;
                    submitBtn.textContent = 'Datei prüfen';
                }
            });
    });

{% if student and student.step_checkboxes and session.student_id %}
/* Arbeitsschritte abhaken -- opt-in per student (student.step_checkboxes).

   Ticks live in localStorage only, never on the server: they are scratchpad
   state for one ~20-minute Aufgabe, and the durable record is student_subtask.
   Keyed by student id because school PCs are shared.

   Steps are keyed by a hash of their own text, not by position. A teacher who
   inserts a step must not shift every tick below it -- that would show a
   student confident green ticks on steps they never did. Hashing instead
   fails empty: fix a typo in a step and its tick quietly disappears. */
(function () {
    const content = document.querySelector('.task-content');
    if (!content) return;

    const steps = Array.from(content.querySelectorAll('ol > li'))
        .filter(li => !li.parentElement.closest('li'));
    if (!steps.length) return;

    const STORE_KEY = 'lm.steps.{{ session.student_id }}.' + topicSlug + '.{{ current_index + 1 }}';

    function stepText(li) {
        // Nested sub-items are excluded so that editing a sub-bullet does not
        // silently drop the tick on the step it belongs to.
        const clone = li.cloneNode(true);
        clone.querySelectorAll('ul, ol, .step-box').forEach(n => n.remove());
        return clone.textContent.replace(/\s+/g, ' ').trim();
    }

    function hash(str) {
        let h = 5381;
        for (let i = 0; i < str.length; i++) h = ((h << 5) + h + str.charCodeAt(i)) | 0;
        return (h >>> 0).toString(36);
    }

    function load() {
        try {
            const raw = localStorage.getItem(STORE_KEY);
            return raw ? new Set(JSON.parse(raw)) : new Set();
        } catch (e) {
            return new Set();
        }
    }

    function save() {
        // Only hashes still present in the DOM are written back, so steps the
        // teacher removed garbage-collect themselves.
        const live = steps.filter(li => li.classList.contains('step-done'))
                          .map(li => li.dataset.stepKey);
        try {
            if (live.length) localStorage.setItem(STORE_KEY, JSON.stringify(live));
            else localStorage.removeItem(STORE_KEY);
        } catch (e) { /* private mode: ticks still work for this page view */ }
    }

    /* The decisive control, in the order the system actually cares about: an
       unpassed artifact gate outranks the completion checkbox, because for MBI
       the gate is what releases the Aufgabe. Ticking every step releases
       nothing -- this is what the nudge has to make obvious. */
    function decisiveTarget() {
        for (const id of ['inline-gate-card', 'gate-card']) {
            const gate = document.getElementById(id);
            if (gate && !gate.classList.contains('gate-passed')) {
                return { el: gate, label: 'Zur Dateiprüfung' };
            }
        }
        const zone = document.getElementById('completion-zone');
        if (zone && !zone.hidden) {
            const cb = zone.querySelector('.subtask-checkbox');
            if (cb && !cb.checked) return { el: zone, label: 'Zum Haken' };
        }
        return null;
    }

    const nudge = document.createElement('div');
    nudge.className = 'step-nudge';
    nudge.hidden = true;
    content.appendChild(nudge);

    function refreshNudge() {
        const allDone = steps.every(li => li.classList.contains('step-done'));
        const target = allDone ? decisiveTarget() : null;
        if (!target) { nudge.hidden = true; return; }

        nudge.innerHTML = '';
        nudge.append('Alle Schritte abgehakt — die Aufgabe selbst ist damit noch nicht fertig.');
        const btn = document.createElement('button');
        btn.type = 'button';
        btn.textContent = target.label;
        btn.addEventListener('click', () => {
            target.el.scrollIntoView({ behavior: 'smooth', block: 'center' });
            target.el.classList.add('step-target-pulse');
            setTimeout(() => target.el.classList.remove('step-target-pulse'), 2500);
        });
        nudge.appendChild(btn);
        nudge.hidden = false;
    }

    function setDone(li, done) {
        li.classList.toggle('step-done', done);
        li.setAttribute('aria-checked', done ? 'true' : 'false');
        li.querySelector('.step-box').textContent = done ? '✓' : '';
        save();
        refreshNudge();
    }

    const stored = load();

    steps.forEach(li => {
        li.dataset.stepKey = hash(stepText(li));
        li.classList.add('step-item');
        li.setAttribute('role', 'checkbox');
        li.setAttribute('tabindex', '0');

        // Row layout is [number] [box] [text]. The step's own content has to
        // live in one element or flex would treat every text node as a column.
        li.parentElement.classList.add('step-list');
        const text = document.createElement('span');
        text.className = 'step-text';
        while (li.firstChild) text.appendChild(li.firstChild);

        const box = document.createElement('span');
        box.className = 'step-box';
        box.setAttribute('aria-hidden', 'true');
        li.appendChild(box);
        li.appendChild(text);

        li.addEventListener('click', ev => {
            // Links, buttons and form controls inside a step keep their own job.
            if (ev.target.closest('a, button, input, textarea, select, label, summary')) return;
            // Highlighting a sentence to read it is not a click.
            if (String(window.getSelection())) return;
            setDone(li, !li.classList.contains('step-done'));
        });

        li.addEventListener('keydown', ev => {
            if (ev.key !== ' ' && ev.key !== 'Enter') return;
            if (ev.target !== li) return;
            ev.preventDefault();
            setDone(li, !li.classList.contains('step-done'));
        });

        if (stored.has(li.dataset.stepKey)) {
            li.classList.add('step-done');
            li.setAttribute('aria-checked', 'true');
            box.textContent = '✓';
        } else {
            li.setAttribute('aria-checked', 'false');
        }
    });

    save();
    refreshNudge();
})();
